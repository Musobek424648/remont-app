function loadObjectExpenses(nomi) { fetch("/api/expenses/" + encodeURIComponent(nomi)) .then(res => res.json()) .then(data => { const tbody = document.getElementById("object-expenses-table-body"); tbody.innerHTML = ""; let totalSum = 0; if (!data || data.length === 0) { tbody.innerHTML = 'Hali xarajat yo`q'; document.getElementById("total-object-expense").innerText = "0 so'm"; return; } data.forEach(exp => { totalSum += Number(exp.summa || 0); tbody.innerHTML += ` ${exp.nomi} ${Number(exp.summa).toLocaleString()} so'm ${exp.created_at ? new Date(exp.created_at).toLocaleString() : ''} `; }); document.getElementById("total-object-expense").innerText = totalSum.toLocaleString() + " so'm"; }); }
